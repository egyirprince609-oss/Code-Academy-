CODE ACADEMY LIVE MAP — FIRESTORE RULES PATCH (SCHEMA v2)

This block must be merged into the project's existing Firestore rules inside:
service cloud.firestore {
  match /databases/{database}/documents {
    ...
  }
}

Do not replace other rules for users, lessons, notifications, profiles, XP,
referrals, community, or any existing collection. Publish the merged rules.

match /mapShares/{shareId} {
  function isSignedInMember() {
    return request.auth != null
      && request.auth.token.firebase.sign_in_provider != 'anonymous';
  }

  function validOptionalText(data, fieldName, maxLength) {
    return data.get(fieldName, null) is string
      && data.get(fieldName, '').size() <= maxLength;
  }

  function validShareFields(data) {
    return data.keys().hasAll([
      'ownerUid', 'ownerName', 'lat', 'lng', 'accuracy', 'address',
      'neighborhood', 'city', 'district', 'region', 'country',
      'countryCode', 'postcode', 'active', 'shareLabel', 'createdAt',
      'updatedAt', 'expiresAt', 'schemaVersion'
    ])
    && data.keys().hasOnly([
      'ownerUid', 'ownerName', 'lat', 'lng', 'accuracy', 'address',
      'neighborhood', 'city', 'district', 'region', 'country',
      'countryCode', 'postcode', 'active', 'shareLabel', 'createdAt',
      'updatedAt', 'expiresAt', 'schemaVersion'
    ])
    && data.ownerName is string && data.ownerName.size() <= 60
    && data.lat is number && data.lat >= -90 && data.lat <= 90
    && data.lng is number && data.lng >= -180 && data.lng <= 180
    && (data.accuracy == null || (data.accuracy is number && data.accuracy >= 0 && data.accuracy <= 100000))
    && validOptionalText(data, 'address', 220)
    && validOptionalText(data, 'neighborhood', 100)
    && validOptionalText(data, 'city', 100)
    && validOptionalText(data, 'district', 100)
    && validOptionalText(data, 'region', 100)
    && validOptionalText(data, 'country', 100)
    && validOptionalText(data, 'countryCode', 8)
    && validOptionalText(data, 'postcode', 30)
    && data.shareLabel is string && data.shareLabel.size() <= 40
    && data.expiresAt is timestamp
    && data.schemaVersion == 2;
  }

  // Direct reads only. A viewer must be signed in, and an active share must not be expired.
  allow get: if isSignedInMember() && (
    resource.data.ownerUid == request.auth.uid
    || (resource.data.active == true
      && resource.data.expiresAt is timestamp
      && resource.data.expiresAt > request.time)
  );
  allow list: if false;

  allow create: if isSignedInMember()
    && shareId.matches('^[a-f0-9]{48}$')
    && validShareFields(request.resource.data)
    && request.resource.data.ownerUid == request.auth.uid
    && request.resource.data.active == true
    && request.resource.data.createdAt == request.time
    && request.resource.data.updatedAt == request.time
    && request.resource.data.expiresAt > request.time
    && request.resource.data.expiresAt <= request.time + duration.value(2, 'h');

  allow update: if isSignedInMember()
    && resource.data.ownerUid == request.auth.uid
    && (
      // Permit a legacy link owner to revoke a pre-v2 share without editing any other field.
      (resource.data.schemaVersion != 2
        && request.resource.data.ownerUid == resource.data.ownerUid
        && request.resource.data.active == false
        && request.resource.data.updatedAt == request.time
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['active', 'updatedAt']))
      ||
      (validShareFields(request.resource.data)
        && request.resource.data.ownerUid == resource.data.ownerUid
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.expiresAt == resource.data.expiresAt
        && request.resource.data.updatedAt == request.time
        && (request.resource.data.active == false
          || (request.resource.data.active == true
            && request.resource.data.expiresAt > request.time
            && request.resource.data.expiresAt <= request.time + duration.value(2, 'h'))))
    );

  allow delete: if isSignedInMember() && resource.data.ownerUid == request.auth.uid;

  // Each authenticated viewer can create a view event. Only the share owner can read them.
  match /views/{viewId} {
    function parentShare() {
      return get(/databases/$(database)/documents/mapShares/$(shareId)).data;
    }

    allow create: if isSignedInMember()
      && viewId.matches('^[A-Za-z0-9_-]{20,128}$')
      && parentShare().active == true
      && parentShare().expiresAt is timestamp
      && parentShare().expiresAt > request.time
      && parentShare().ownerUid != request.auth.uid
      && request.resource.data.keys().hasAll(['viewerUid', 'viewerName', 'viewedAt', 'schemaVersion'])
      && request.resource.data.keys().hasOnly(['viewerUid', 'viewerName', 'viewedAt', 'schemaVersion'])
      && request.resource.data.viewerUid == request.auth.uid
      && request.resource.data.viewerName is string
      && request.resource.data.viewerName.size() <= 60
      && request.resource.data.viewedAt == request.time
      && request.resource.data.schemaVersion == 1;

    allow get, list: if isSignedInMember() && parentShare().ownerUid == request.auth.uid;
    allow update, delete: if false;
  }
}

SECURITY AND PRODUCT NOTES

- Location links require a non-anonymous Firebase account. The viewer's account
  display name may appear in the owner's view alert.
- A random 48-character share ID is not a replacement for authentication. Keep
  location links private and revoke them when they are no longer needed.
- Share documents expire after two hours, and owners can stop them sooner.
- The owner receives real-time in-app/account alerts when the owner page is open.
  View events remain in Firestore and can be processed when the owner returns.
  Immediate device push while every owner page is closed requires a configured
  server-side push delivery system; a browser page alone cannot guarantee it.
- These rules validate data access, but existing rules for the app's account
  notifications must also allow a user to write only to their own notification
  subcollection through the existing notification engine.
