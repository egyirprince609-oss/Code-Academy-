1. Personal Portfolio Page

EASY

Your first website. Introduce yourself with photo, bio, and contact links.

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>My Portfolio</title>
</head>
<body>
    <header>
        <h1>John Doe</h1>
        <p>Web Developer Student</p>
    </header>
    
    <nav>
        <a href="#about">About</a> | 
        <a href="#skills">Skills</a> | 
        <a href="#contact">Contact</a>
    </nav>
    
    <main>
        <section id="about">
            <h2>About Me</h2>
            <p>Hello! I am learning HTML at Code Academy.</p>
            <img src="profile.jpg" alt="My photo" width="150">
        </section>
        
        <section id="skills">
            <h2>My Skills</h2>
            <ul>
                <li>HTML Structure</li>
                <li>Creating Lists</li>
                <li>Adding Links</li>
            </ul>
        </section>
        
        <section id="contact">
            <h2>Contact Me</h2>
            <p>Email: john@email.com</p>
            <a href="https://github.com">GitHub</a>
        </section>
    </main>
    
    <footer>
        <p>Copyright © 2026 John Doe</p>
    </footer>
</body>
</html>
