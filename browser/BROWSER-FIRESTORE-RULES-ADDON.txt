# Each signed-in user may read/write only their own browser profile and its browser data.
match /browserProfiles/{uid} {
  allow read, create, update, delete: if request.auth != null && request.auth.uid == uid;
  match /browserTabs/{tabId} {
    allow read, create, update, delete: if request.auth != null && request.auth.uid == uid;
  }
  match /browserHistory/{entryId} {
    allow read, create, update, delete: if request.auth != null && request.auth.uid == uid;
  }
  match /browserSettings/{settingId} {
    allow read, create, update, delete: if request.auth != null && request.auth.uid == uid;
  }
  match /notifications/{notificationId} {
    allow read, create, update, delete: if request.auth != null && request.auth.uid == uid;
  }
}
