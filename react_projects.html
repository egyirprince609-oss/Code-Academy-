import React, { useState, useEffect } from 'react'; function GuessingGame() { const [number, setNumber] = useState(0); const [guess, setGuess] = useState(''); const [message, setMessage] = useState('Start guessing!'); const [attempts, setAttempts] = useState(0); useEffect(() => { setNumber(Math.floor(Math.random() * 100) + 1); }, []); const makeGuess = () => { const numGuess = parseInt(guess); setAttempts(attempts + 1); if(numGuess === number) { setMessage('🎉 Correct! You won!'); } else if(numGuess < number) { setMessage('📈 Too Low! Try higher'); } else { setMessage('📉 Too High! Try lower'); } setGuess(''); }; const resetGame = () => { setNumber(Math.floor(Math.random() * 100) + 1); setAttempts(0); setMessage('Start guessing!'); setGuess(''); }; return ( <div> <h3>Guess 1-100</h3> <input type="number" value={guess} onChange={(e) => setGuess(e.target.value)} /> <button onClick={makeGuess}>Guess</button> <button onClick={resetGame}>New Game</button> <p>{message}</p> <p>Attempts: {attempts}</p> </div> ); } export default GuessingGame;
